cmake_minimum_required(VERSION 3.16)
project(ptz_onvif LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(${CMAKE_CURRENT_SOURCE_DIR}/../../common/config/edge_config.cmake)
find_package(EXPAT REQUIRED)
find_package(OpenSSL REQUIRED)
find_path(UUID_INCLUDE_DIR uuid/uuid.h REQUIRED)
find_library(UUID_LIBRARY uuid REQUIRED)
add_library(onvif_protocol src/xml/xml.cpp src/protocol/service.cpp src/auth/digest.cpp src/ptz/ptz_service.cpp src/ptz/servo_bridge.cpp src/events/event_broker.cpp src/events/event_service.cpp src/control/camera_control.cpp)
target_include_directories(onvif_protocol PUBLIC src PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../common)
target_link_libraries(onvif_protocol PUBLIC edge_config EXPAT::EXPAT OpenSSL::Crypto)
target_compile_options(onvif_protocol PRIVATE -Wall -Wextra -Wpedantic)
add_executable(ptz_onvif src/main.cpp src/network/server.cpp src/events/servo_feed.cpp)
target_link_libraries(ptz_onvif PRIVATE onvif_protocol ${UUID_LIBRARY})
target_include_directories(ptz_onvif PRIVATE ${UUID_INCLUDE_DIR})
target_compile_options(ptz_onvif PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(protocol_test tests/protocol_test.cpp)
target_link_libraries(protocol_test PRIVATE onvif_protocol)
target_compile_options(protocol_test PRIVATE -UNDEBUG)
add_test(NAME protocol COMMAND protocol_test)
add_executable(network_test tests/network_test.cpp)
target_link_libraries(network_test PRIVATE onvif_protocol)
target_compile_options(network_test PRIVATE -UNDEBUG)
add_test(NAME http_network COMMAND network_test $<TARGET_FILE:ptz_onvif>)
set_tests_properties(http_network PROPERTIES TIMEOUT 15)
find_package(CURL REQUIRED)
add_library(onvif_client tools/soap_client.cpp)
target_include_directories(onvif_client PUBLIC tools)
target_link_libraries(onvif_client PUBLIC onvif_protocol CURL::libcurl)
add_executable(onvif_query tools/query_stream.cpp)
target_link_libraries(onvif_query PRIVATE onvif_client)
add_executable(onvif_probe tools/probe.cpp)
target_link_libraries(onvif_probe PRIVATE onvif_protocol)
add_executable(configure_access tools/configure_access.cpp)
target_link_libraries(configure_access PRIVATE onvif_protocol)
add_executable(authentication_test tests/authentication_test.cpp)
target_link_libraries(authentication_test PRIVATE onvif_protocol CURL::libcurl)
target_compile_options(authentication_test PRIVATE -UNDEBUG)
add_test(NAME authentication COMMAND authentication_test $<TARGET_FILE:ptz_onvif>)
set_tests_properties(authentication PROPERTIES TIMEOUT 15)
add_executable(check_rtsp_auth tools/check_rtsp_auth.cpp)
target_link_libraries(check_rtsp_auth PRIVATE onvif_protocol)
add_executable(ptz_test tests/ptz_test.cpp)
target_link_libraries(ptz_test PRIVATE onvif_protocol)
target_compile_options(ptz_test PRIVATE -UNDEBUG)
add_test(NAME ptz_protocol COMMAND ptz_test)
add_executable(onvif_ptz_check tools/ptz_check.cpp)
target_link_libraries(onvif_ptz_check PRIVATE onvif_client)
add_executable(events_test tests/events_test.cpp)
target_link_libraries(events_test PRIVATE onvif_protocol)
target_include_directories(events_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../common)
target_compile_options(events_test PRIVATE -UNDEBUG)
add_test(NAME events COMMAND events_test)
add_executable(onvif_events_check tools/events_check.cpp)
target_link_libraries(onvif_events_check PRIVATE onvif_client)
find_package(Threads REQUIRED)
add_executable(events_network_test tests/events_network_test.cpp)
target_link_libraries(events_network_test PRIVATE onvif_client Threads::Threads)
target_compile_options(events_network_test PRIVATE -UNDEBUG)
add_test(NAME events_network COMMAND events_network_test $<TARGET_FILE:ptz_onvif>)
set_tests_properties(events_network PROPERTIES TIMEOUT 15)

add_executable(camera_control_test tests/camera_control_test.cpp)
target_link_libraries(camera_control_test PRIVATE onvif_protocol)
target_compile_options(camera_control_test PRIVATE -UNDEBUG)
add_test(NAME camera_control COMMAND camera_control_test)
